cmake_minimum_required(VERSION 3.16)
project(plamo_browser LANGUAGES C CXX)

set(GGML_WEBGPU ON CACHE BOOL "")
set(GGML_OPENMP OFF CACHE BOOL "")
set(LLAMA_BUILD_HTML OFF CACHE BOOL "")
set(LLAMA_WASM_MEM64 OFF CACHE BOOL "")
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "")
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "")
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "")
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "")
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "")
set(LLAMA_OPENSSL OFF CACHE BOOL "")
if(NOT GGML_WEBGPU)
    add_compile_options("-msimd128")
endif()
add_subdirectory("${LLAMA_CPP_DIR}" llama)
configure_file("${LLAMA_CPP_DIR}/LICENSE" llama-license.txt COPYONLY)

add_executable(plamo inference.cpp)
target_link_libraries(plamo PRIVATE llama)
if(LLAMA_WASM_MEM64)
    target_compile_options(plamo PRIVATE "-sMEMORY64=1")
    target_link_options(plamo PRIVATE "-sMEMORY64=1" "-sMAXIMUM_MEMORY=17179869184")
else()
    target_link_options(plamo PRIVATE "-sMAXIMUM_MEMORY=4294967296")
endif()
target_link_options(plamo PRIVATE
    "-O1" "--no-entry" "-lworkerfs.js" "-sMODULARIZE=1" "-sEXPORT_ES6=1"
    "-sENVIRONMENT=worker" "-sALLOW_MEMORY_GROWTH=1" "-sFORCE_FILESYSTEM=1" "-sSTACK_SIZE=1048576"
    "-sEXPORTED_FUNCTIONS=_malloc,_free"
    "-sEXPORTED_RUNTIME_METHODS=ccall,FS,WORKERFS,HEAPU8")
if(GGML_WEBGPU)
    target_link_options(plamo PRIVATE "-sJSPI_EXPORTS=load_model,start_translation,next_token")
endif()
set_target_properties(plamo PROPERTIES SUFFIX ".mjs")
